#include <xc.h>
#include <libpic30.h>
#include "buttons.h"  
#include "adc.h"      
 
int main(void) {
    TRISA = 0x0000;
   
    ADC_SetConfiguration(ADC_CONFIGURATION_DEFAULT);
    ADC_ChannelEnable(ADC_CHANNEL_POTENTIOMETER);
 
    unsigned int wartosc_potencjometru = 0;
   
    int alarm_wlaczony = 0;
    int czas_mrugania = 0;
 
    while(1) {
        wartosc_potencjometru = ADC_Read10bit(ADC_CHANNEL_POTENTIOMETER);
       
        if (wartosc_potencjometru == 0xFFFF) {
            continue;
        }
 
        if (alarm_wlaczony == 0) {
            if (wartosc_potencjometru > 512) {
                alarm_wlaczony = 1;
                czas_mrugania = 0;  
            }
        }
 
        if (alarm_wlaczony == 1) {
            if (wartosc_potencjometru <= 512) {
                if (BUTTON_IsPressed(BUTTON_S3) == true) {
                    alarm_wlaczony = 0;
                }
            }
        }
 
        if (alarm_wlaczony == 0) {
            LATA = 0x00;
        }
        else if (alarm_wlaczony == 1) {
           
            if (czas_mrugania < 10) {
               
                if (LATA == 0x00) {
                    LATA = 0x01;
                } else {
                    LATA = 0x00;
                }
               
                czas_mrugania = czas_mrugania + 1;
               
            }
            else {
                LATA = 0xFF;
            }
           
        }
 
        _delay32(1500000);
    }
   
    return 0;
}
