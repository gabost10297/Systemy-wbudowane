#include <xc.h>

#include <libpic30.h>

#include <stdio.h>

#include <string.h>

#include "lcd.h"

#include "buttons.h"

int main(void) {

    LCD_Initialize();

    int czas_gracza1 = 180;

    int czas_gracza2 = 180;

    int zegar1_tyka = 0;

    int zegar2_tyka = 0;
 
    int stoper = 0;

    char tekst_na_ekran[17];

    while(1) {
 
        if (BUTTON_IsPressed(BUTTON_S3) == true) {

            if (czas_gracza1 > 0 && czas_gracza2 > 0) {

                zegar1_tyka = 0;

                zegar2_tyka = 1;

            }

        }

        if (BUTTON_IsPressed(BUTTON_S4) == true) {

            if (czas_gracza1 > 0 && czas_gracza2 > 0) {

                zegar1_tyka = 1;

                zegar2_tyka = 0;

            }

        }


        stoper = stoper + 1;

        if (stoper == 10) {

            if (zegar1_tyka == 1) {

                czas_gracza1 = czas_gracza1 - 1;

            }

            if (zegar2_tyka == 1) {

                czas_gracza2 = czas_gracza2 - 1;

            }

            stoper = 0;

        }

        if (czas_gracza1 == 0 || czas_gracza2 == 0) {

            zegar1_tyka = 0;

            zegar2_tyka = 0;

        }

        LCD_ClearScreen();

        if (czas_gracza1 == 0) {

            sprintf(tekst_na_ekran, "G1 PRZEGRAL CZAS");

        } 

        else if (czas_gracza2 == 0) {

            sprintf(tekst_na_ekran, "G2 PRZEGRAL CZAS");

        } 

        else {
 
            int min1 = czas_gracza1 / 60;

            int sek1 = czas_gracza1 % 60;

            int min2 = czas_gracza2 / 60;

            int sek2 = czas_gracza2 % 60;
 
            sprintf(tekst_na_ekran, "G1 %02u:%02u G2 %02u:%02u", min1, sek1, min2, sek2);

        }

        LCD_PutString(tekst_na_ekran, strlen(tekst_na_ekran));
 
        __delay32(400000);

    }

    return 0;

}
