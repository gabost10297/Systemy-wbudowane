#include <xc.h>
#include <libpic30.h>
#include <stdio.h>
#include <string.h>
#include "lcd.h" 
#include "buttons.h"
#include "adc.h" 
 
int main(void) {
    ADC_SetConfiguration(ADC_CONFIGURATION_DEFAULT); 
    ADC_ChannelEnable(ADC_CHANNEL_POTENTIOMETER);   
    LCD_Initialize();     

    int czas = 0;           
    int moc = 0;            
    int mikrofala_dziala = 0;
    int stoper = 0;    
 
    char bufor[17];
 
    while(1) {
        // Wybor mocy
        if (mikrofala_dziala == 0) {
            unsigned int adc_val = ADC_Read10bit(ADC_CHANNEL_POTENTIOMETER); 
            if (adc_val != 0xFFFF) { 
                moc = (adc_val * 100) / 1023; 
            }
        }
 
        // OBSŁUGA PRZYCISKÓW
        if (BUTTON_IsPressed(BUTTON_S3) == true) {
            czas = czas + 30;
            _delay32(1500000); 
        }
 
        // Przycisk START / STOP
        if (BUTTON_IsPressed(BUTTON_S4) == true) {
            if (czas > 0) { 
                if (mikrofala_dziala == 0) {
                    mikrofala_dziala = 1;
                } else {
                    mikrofala_dziala = 0; 
                }
            }
            _delay32(1500000);
        }
 
        // RESET
        if (BUTTON_IsPressed(BUTTON_S5) == true) {
            mikrofala_dziala = 0;
            czas = 0;
            _delay32(1500000);
        }
 
        // ODLICZANIE CZASU
        if (mikrofala_dziala == 1) {
            stoper++;
            if (stoper >= 10) {
                czas--;
                stoper = 0;
            }
 
            
            if (czas <= 0) {
                mikrofala_dziala = 0;
                czas = 0;
            }
        }
 
        // WYŚWIETLANIE NA EKRANIE LCD
        unsigned char minuty = czas / 60;
        unsigned char sekundy = czas % 60;

        if (mikrofala_dziala == 1) {
            sprintf(bufor, "RUN  %02u:%02u %3d%%", minuty, sekundy, moc);
        } else {
            sprintf(bufor, "STOP %02u:%02u %3d%%", minuty, sekundy, moc);
        }

        LCD_ClearScreen();
        LCD_PutString(bufor, strlen(bufor));

        __delay32(400000); 
    }
 
    retu
